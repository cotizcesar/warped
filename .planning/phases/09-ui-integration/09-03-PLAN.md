**Part A — EngineManager.kt: cacheDir caching**

     companion object {

2. **Add private helper method for cache path resolution:**
   ```kotlin
   private fun getCachedModelPath(originalPath: String): File {
       val cacheDir = File(context.cacheDir, CACHE_SUBDIR)
       if (!cacheDir.exists()) cacheDir.mkdirs()
       // Use the filename from originalPath to avoid collisions
       val fileName = File(originalPath).name
       return File(cacheDir, fileName)
   }
   ```
   Add import: `import java.io.File`

3. **Modify switchToLiteRT() to use cached copy:**
   ```kotlin
   @Synchronized
   fun switchToLiteRT(modelPath: String) {
       val cachedFile = getCachedModelPath(modelPath)
       
       // If cached copy doesn't exist, copy the model to cacheDir
       if (!cachedFile.exists()) {
           Timber.d("EngineManager: caching .litertlm model to ${cachedFile.absolutePath}")
           try {
               val source = File(modelPath)
               source.copyTo(cachedFile, overwrite = false)
               Timber.d("EngineManager: model cached successfully (${cachedFile.length()} bytes)")
           } catch (e: Exception) {
               Timber.w(e, "EngineManager: failed to cache model, using original path")
               // Fall through — use original path if cache fails
           }
       }
       
       val resolvedPath = if (cachedFile.exists()) cachedFile.absolutePath else modelPath
       val target = ActiveEngine(EngineType.LITE_RT_LM, resolvedPath, backendDetector.probeBackend())
       if (activeEngine == target) {
           Timber.d("EngineManager: $target already loaded, skipping switch")
           return
       }
       unloadCurrent()
       
       Timber.d("EngineManager: initializing LiteRT-LM with backend=${target.backend} path=$resolvedPath")
       liteRTLmEngine.init(resolvedPath, target.backend!!)
       activeEngine = target
       Timber.d("EngineManager: LiteRT-LM engine now active")
   }
   ```

   Key design decisions:
   - Cache on first init only (copy if not exists)
   - On subsequent loads, reuse cached copy (fast path — no copy needed)
   - If cache copy fails (disk full, permissions), fall back to original path
   - ActiveEngine stores the **resolved** path for deduplication
   - The copy is a simple `File.copyTo()` which handles large files via buffered streams
   - Cache directory is `context.cacheDir/litertlm_cache/` — survives app restarts within same app version but may be cleared by system under storage pressure (acceptable per CONTEXT.md: "Cache does NOT survive app restart")

4. **Add handleTrimMemory() method:**
   ```kotlin
   /**
    * Handle system memory pressure. Called from Application.onTrimMemory.
    * Releases engine resources on critical memory pressure.
    */
   fun handleTrimMemory(level: Int) {
       if (level >= android.content.ComponentCallbacks2.TRIM_MEMORY_RUNNING_CRITICAL) {
           Timber.d("EngineManager: TRIM_MEMORY_RUNNING_CRITICAL — unloading engine")
           try {
               unloadCurrent()
               // Also clear cache to free disk space
               val cacheDir = File(context.cacheDir, CACHE_SUBDIR)
               if (cacheDir.exists()) {
                   cacheDir.deleteRecursively()
                   Timber.d("EngineManager: cache directory cleared")
               }
           } catch (e: Exception) {
               Timber.w(e, "EngineManager: error during trim memory")
           }
       }
   }
   ```
   - Clears both engine memory (unloadCurrent) and cache directory on critical trim
   - The next load after trim will recopy from source to cacheDir
   - Only acts on TRIM_MEMORY_RUNNING_CRITICAL (level 15) — lower levels (UI_HIDDEN, BACKGROUND) are ignored since the user may return quickly
