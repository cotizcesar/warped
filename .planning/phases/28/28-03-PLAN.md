package com.warped.data.highlighting

import com.google.common.truth.Truth.assertThat
import com.warped.domain.model.TokenType
import dev.snipme.highlights.model.CodeStructure
import dev.snipme.highlights.model.PhraseLocation
import org.junit.jupiter.api.Test

class TypeMapperTest {

    // Test 1: Simple Java snippet — keyword + string + punctuation + marks
    @Test
    fun `maps keywords to KEYWORD type`() {
        val code = "public class Foo {}"
        val structure = CodeStructure(
            marks = setOf(PhraseLocation(16, 17), PhraseLocation(18, 19)), // {, }
            punctuations = emptySet(),
            keywords = setOf(PhraseLocation(0, 6), PhraseLocation(7, 12)), // public, class
            strings = emptySet(),
            literals = emptySet(),
            comments = emptySet(),
            multilineComments = emptySet(),
            annotations = emptySet(),
            incremental = false,
        )
        val tokens = TypeMapper.map(structure, code)
        val keywordTokens = tokens.filter { it.type == TokenType.KEYWORD }
        assertThat(keywordTokens).hasSize(2)
        assertThat(keywordTokens[0].text).isEqualTo("public")
        assertThat(keywordTokens[1].text).isEqualTo("class")
    }

    // Test 2: String token detection
    @Test
    fun `maps strings to STRING type`() {
        val code = "val x = \"hello world\""
        val structure = CodeStructure(
            marks = emptySet(),
            punctuations = setOf(PhraseLocation(6, 7)), // =
            keywords = setOf(PhraseLocation(0, 3)),     // val
            strings = setOf(PhraseLocation(8, 21)),      // "hello world"
            literals = emptySet(),
            comments = emptySet(),
            multilineComments = emptySet(),
            annotations = emptySet(),
            incremental = false,
        )
        val tokens = TypeMapper.map(structure, code)
        val stringTokens = tokens.filter { it.type == TokenType.STRING }
        assertThat(stringTokens).hasSize(1)
        assertThat(stringTokens[0].text).isEqualTo("\"hello world\"")
    }

    // Test 3: Gap-filling — every character position is covered
    @Test
    fun `fills all character positions with no gaps`() {
        val code = "val x = 42"
        val structure = CodeStructure(
            marks = emptySet(),
            punctuations = setOf(PhraseLocation(6, 7)), // =
            keywords = setOf(PhraseLocation(0, 3)),     // val
            strings = emptySet(),
            literals = setOf(PhraseLocation(8, 10)),     // 42
            comments = emptySet(),
            multilineComments = emptySet(),
            annotations = emptySet(),
            incremental = false,
        )
        val tokens = TypeMapper.map(structure, code)
        // Verify coverage
        var pos = 0
        for (token in tokens) {
            assertThat(token.start).isEqualTo(pos)
            pos = token.end
        }
        assertThat(pos).isEqualTo(code.length)
    }

    // Test 4: Literal splitting — numeric vs constant
    @Test
    fun `splits literals into NUMBER and CONSTANT`() {
        val code = "val a = 42; val b = true; val c = null"
        val structure = CodeStructure(
            marks = emptySet(),
            punctuations = setOf(PhraseLocation(6, 7), PhraseLocation(9, 10), PhraseLocation(21, 22), PhraseLocation(24, 25), PhraseLocation(36, 37)),
            keywords = setOf(PhraseLocation(0, 3), PhraseLocation(11, 14), PhraseLocation(26, 29)),
            strings = emptySet(),
            literals = setOf(PhraseLocation(8, 10), PhraseLocation(23, 27), PhraseLocation(38, 42)), // 42, true, null
            comments = emptySet(),
            multilineComments = emptySet(),
            annotations = emptySet(),
            incremental = false,
        )
        val tokens = TypeMapper.map(structure, code)
        val numberTokens = tokens.filter { it.type == TokenType.NUMBER }
        val constantTokens = tokens.filter { it.type == TokenType.CONSTANT }
        assertThat(numberTokens).hasSize(1)
        assertThat(numberTokens[0].text).isEqualTo("42")
        assertThat(constantTokens).hasSize(2)
        assertThat(constantTokens.map { it.text }).containsExactly("true", "null")
    }

    // Test 5: Comment detection
    @Test
    fun `maps comments to COMMENT type`() {
        val code = "// this is a comment\nval x = 1"
        val structure = CodeStructure(
            marks = emptySet(),
            punctuations = setOf(PhraseLocation(26, 27)), // =
            keywords = setOf(PhraseLocation(20, 23)),     // val
            strings = emptySet(),
            literals = setOf(PhraseLocation(28, 29)),      // 1
            comments = setOf(PhraseLocation(0, 19)),       // // this is a comment
            multilineComments = emptySet(),
            annotations = emptySet(),
            incremental = false,
        )
        val tokens = TypeMapper.map(structure, code)
        val commentTokens = tokens.filter { it.type == TokenType.COMMENT }
        assertThat(commentTokens).hasSize(1)
        assertThat(commentTokens[0].text).startsWith("//")
    }

    // Test 6: Punctuation splitting — operator vs punctuation
    @Test
    fun `splits punctuations into OPERATOR and PUNCTUATION`() {
        val code = "x = (a + b) * c"
        val structure = CodeStructure(
            marks = emptySet(),
            punctuations = setOf(
                PhraseLocation(2, 3),  // =
                PhraseLocation(4, 5),  // (
                PhraseLocation(8, 9),  // +
                PhraseLocation(12, 13), // )
                PhraseLocation(14, 15), // *
            ),
            keywords = emptySet(),
            strings = emptySet(),
            literals = emptySet(),
            comments = emptySet(),
            multilineComments = emptySet(),
            annotations = emptySet(),
            incremental = false,
        )
        val tokens = TypeMapper.map(structure, code)
        val operatorTokens = tokens.filter { it.type == TokenType.OPERATOR }
        val punctuationTokens = tokens.filter { it.type == TokenType.PUNCTUATION }
        assertThat(operatorTokens.map { it.text }).containsExactly("=", "+", "*")
        assertThat(punctuationTokens.map { it.text }).containsExactly("(", ")")
    }

    // Test 7: Empty code
    @Test
    fun `handles empty code gracefully`() {
        val code = ""
        val structure = CodeStructure(
            marks = emptySet(), punctuations = emptySet(), keywords = emptySet(),
            strings = emptySet(), literals = emptySet(), comments = emptySet(),
            multilineComments = emptySet(), annotations = emptySet(), incremental = false,
        )
        val tokens = TypeMapper.map(structure, code)
        assertThat(tokens).isEmpty()
    }

    // Test 8: Single character code
    @Test
    fun `handles single character code`() {
        val code = "x"
        val structure = CodeStructure(
            marks = emptySet(), punctuations = emptySet(), keywords = emptySet(),
            strings = emptySet(), literals = emptySet(), comments = emptySet(),
            multilineComments = emptySet(), annotations = emptySet(), incremental = false,
        )
        val tokens = TypeMapper.map(structure, code)
