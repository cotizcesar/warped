### `app/src/main/java/com/warped/domain/model/AllowlistEntry.kt`
```kotlin
@Serializable
data class AllowlistEntry(
    val name: String,
    val displayName: String,
    val modelFile: String,
    val sizeInBytes: Long,
    val capabilities: List<String> = emptyList(),
    val llmPromptTemplates: Map<String, String> = emptyMap(),
    val taskTypes: List<String> = emptyList(),
) {
    val id: String get() = name  // alias for lookups
    fun hasCapability(cap: String): Boolean = cap in capabilities
}

@Serializable
data class ModelAllowlist(
    val models: List<AllowlistEntry> = emptyList(),
)
```

### `app/src/main/java/com/warped/domain/repository/ModelAllowlistRepository.kt`
```kotlin
interface ModelAllowlistRepository {
    suspend fun getAll(): List<AllowlistEntry>
    fun observeAll(): Flow<List<AllowlistEntry>>
    fun findById(modelId: String): AllowlistEntry?
    fun supportsThinking(modelId: String): Boolean
    fun supportsSpeculativeDecoding(modelId: String): Boolean
    fun supportsVision(modelId: String): Boolean
}
```

### `app/src/main/java/com/warped/data/repository/ModelAllowlistRepositoryImpl.kt`
- `@Singleton class ModelAllowlistRepositoryImpl @Inject constructor(@ApplicationContext context: Context, json: Json) : ModelAllowlistRepository`.
- Reads `assets/model_allowlist.json` on init via `context.assets.open(...)`, parses with kotlinx-serialization `Json.decodeFromString<ModelAllowlist>(...)`.
- Caches result in `private val entries: List<AllowlistEntry>` initialized lazily in init block.
- `observeAll()` returns the cached list wrapped in `MutableStateFlow`.
- `findById(modelId)` matches by `name` (full HF id) or by substring of `modelFile` (handles user-selected local model where id is a path).
- `supportsThinking(modelId)` returns `findById(modelId)?.hasCapability("llm_thinking") == true`.
- `supportsSpeculativeDecoding(modelId)` returns `findById(modelId)?.hasCapability("llm_spec_decoding") == true`.
- `supportsVision(modelId)` returns `findById(modelId)?.hasCapability("llm_vision") == true`.
