### 2. ViewModel
```kotlin
@HiltViewModel
class BenchmarkViewModel @Inject constructor(
    private val localModelRepo: LocalModelRepository,
    private val benchmarkRepo: BenchmarkRepository,
    private val scheduler: BenchmarkScheduler,
    private val workManager: WorkManager,
) : ViewModel() {
    private val _config = MutableStateFlow(BenchmarkConfig())
    private val _selected = MutableStateFlow<LocalModel?>(null)
    val uiState: StateFlow<BenchmarkUiState> = combine(
        localModelRepo.observeDownloaded(),
        _selected, _config,
        benchmarkRepo.observeAll(),
        workManager.getWorkInfosForUniqueWorkLiveData(ModelBenchmarkWorker.UNIQUE_NAME).asFlow(),
    ) { models, selected, config, results, workInfos ->
        val running = workInfos.any { it.state == WorkInfo.State.RUNNING }
        BenchmarkUiState(
            downloadedModels = models,
            selectedModel = selected ?: models.firstOrNull(),
            config = config,
            results = results,
            isRunning = running,
        )
    }.stateIn(viewModelScope, SharingStarted.WhileSubscribed(5_000), BenchmarkUiState())

    fun select(m: LocalModel) { _selected.value = m }
    fun setTemperature(v: Float) { _config.update { it.copy(temperature = v) } }
    fun setTopK(v: Int) { _config.update { it.copy(topK = v) } }
    fun setMaxTokens(v: Int) { _config.update { it.copy(maxTokens = v) } }
    fun setTrials(v: Int) { _config.update { it.copy(trials = v.coerceIn(1, 10)) } }

    fun start() {
        val m = _selected.value ?: return
        scheduler.enqueue(m.filePath, m.name, _config.value)
    }
}
```

```kotlin
@Composable
fun BenchmarkScreen(viewModel: BenchmarkViewModel = hiltViewModel()) {
    val ui by viewModel.uiState.collectAsStateWithLifecycle()
    Scaffold(
        topBar = { TopAppBar(title = { Text("Benchmark") }) },
    ) { pv ->
        Column(modifier = Modifier.padding(pv).padding(16.dp).verticalScroll(rememberScrollState())) {
            Text("Model", style = MaterialTheme.typography.labelLarge)
            ModelDropdown(
                models = ui.downloadedModels,
                selected = ui.selectedModel,
                onSelect = viewModel::select,
            )
            Spacer(Modifier.height(16.dp))
            Text("Configuration", style = MaterialTheme.typography.labelLarge)
            BenchmarkConfigCard(
                config = ui.config,
                onTemperature = viewModel::setTemperature,
                onTopK = viewModel::setTopK,
                onMaxTokens = viewModel::setMaxTokens,
                onTrials = viewModel::setTrials,
            )
            Spacer(Modifier.height(16.dp))
            Button(
                onClick = viewModel::start,
                enabled = !ui.isRunning && ui.selectedModel != null,
                modifier = Modifier.fillMaxWidth(),
            ) {
                if (ui.isRunning) {
                    CircularProgressIndicator(modifier = Modifier.size(16.dp), strokeWidth = 2.dp)
                    Spacer(Modifier.width(8.dp))
                    Text("Running…")
                } else {
                    Icon(Icons.Filled.PlayArrow, null)
                    Spacer(Modifier.width(8.dp))
                    Text("Start Benchmark")
                }
            }
            Spacer(Modifier.height(24.dp))
            Text("Recent results", style = MaterialTheme.typography.labelLarge)
            BenchmarkResultsViewer(results = ui.results)
        }
    }
}
```

```kotlin
@Composable
fun BenchmarkResultsViewer(results: List<BenchmarkResult>) {
    if (results.isEmpty()) {
        Box(Modifier.fillMaxWidth().padding(32.dp), Alignment.Center) {
            Text("No benchmarks yet — run one above.",
                 color = MaterialTheme.colorScheme.onSurfaceVariant)
        }
        return
    }
    LazyColumn(verticalArrangement = Arrangement.spacedBy(8.dp)) {
        items(results, key = { it.id }) { r ->
            Card(modifier = Modifier.fillMaxWidth()) {
                Column(modifier = Modifier.padding(12.dp)) {
                    Row(horizontalArrangement = Arrangement.SpaceBetween,
                        modifier = Modifier.fillMaxWidth()) {
                        Text(r.modelId, style = MaterialTheme.typography.titleMedium)
                        Text(formatTime(r.createdAt),
                             style = MaterialTheme.typography.bodySmall,
                             color = MaterialTheme.colorScheme.onSurfaceVariant)
                    }
                    Spacer(Modifier.height(4.dp))
                    Text("Init: ${r.initTimeMs}ms  Prefill: ${"%.1f".format(r.prefillTokPerSec)} tok/s  " +
                         "Decode: ${"%.1f".format(r.decodeTokPerSec)} tok/s",
                         style = MaterialTheme.typography.bodyMedium)
                    Text("Peak mem: ${formatBytes(r.peakMemoryBytes)}",
                         style = MaterialTheme.typography.bodySmall)
                    Spacer(Modifier.height(4.dp))
                    BenchmarkValueSeriesViewer(
                        values = results.filter { it.modelId == r.modelId }
                            .take(10).map { it.decodeTokPerSec.toFloat() },
                    )
                }
            }
        }
    }
}
```

### 7. BenchmarkValueSeriesViewer (sparkline)
```kotlin
@Composable
fun BenchmarkValueSeriesViewer(values: List<Float>, modifier: Modifier = Modifier) {
    if (values.size < 2) return
    val color = MaterialTheme.colorScheme.primary
    Canvas(modifier = modifier.size(width = 80.dp, height = 24.dp)) {
        val min = values.min(); val max = values.max()
        val range = (max - min).coerceAtLeast(0.001f)
        val stepX = size.width / (values.size - 1)
        val path = Path().apply {
            values.forEachIndexed { i, v ->
                val x = i * stepX
                val y = size.height - ((v - min) / range) * size.height
                if (i == 0) moveTo(x, y) else lineTo(x, y)
            }
        }
        drawPath(path, color = color, style = Stroke(width = 2.dp.toPx(), cap = StrokeCap.Round))
    }
}
```
