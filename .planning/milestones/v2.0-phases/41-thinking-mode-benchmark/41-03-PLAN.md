### 6. Migration test
```kotlin
@RunWith(AndroidJUnit4::class)
class MigrationTest {
    @get:Rule
    val helper = MigrationTestHelper(
        InstrumentationRegistry.getInstrumentation(),
        AppDatabase::class.java,
    )

    @Test
    fun migrate11To12_preservesConversationsAndMessages() {
        // 1. Open v11 schema, insert 1 conversation + 50 messages
        helper.createDatabase(TEST_DB, 11).use { db ->
            db.execSQL("INSERT INTO conversations(...) VALUES(...)")
            repeat(50) { i ->
                db.execSQL("INSERT INTO messages(...) VALUES($i, ...)")
            }
        }
        // 2. Run migration to v12 (autoMigration)
        helper.runMigrationsAndValidate(TEST_DB, 12, true).use { db ->
            val cur = db.query("SELECT COUNT(*) FROM messages WHERE conversationId = 1")
            cur.moveToFirst()
            assertEquals(50, cur.getInt(0))
        }
    }

    private companion object { const val TEST_DB = "migration-test.db" }
}
```
