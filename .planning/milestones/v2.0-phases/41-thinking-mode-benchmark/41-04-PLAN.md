# Plan 41-04: Benchmark — Worker (BENCH-02)

**Phase:** 41 — Thinking Mode + Model Benchmark
**Requirements:** BENCH-02

## Goal

Ship `ModelBenchmarkWorker` that runs a benchmark in the background via WorkManager with `setForeground()` for the long-running notification + `Constraints(UNMETERED, BATTERY_NOT_LOW)`. Worker measures init time, prefill tok/s, decode tok/s, and peak memory across N trials and persists results to `BenchmarkRepository`.

## Scope

- `app/src/main/java/com/warped/data/local/benchmark/ModelBenchmarkWorker.kt` (new `@HiltWorker`)
- `app/src/main/java/com/warped/data/local/benchmark/BenchmarkScheduler.kt` (new — wraps `WorkManager.enqueueUniqueWork()`)
- `app/src/main/assets/benchmark_prompt.txt` (new — ~512-token fixed corpus for repeatable measurement)
- `app/src/main/java/com/warped/data/local/benchmark/MemorySampler.kt` (new — polls native + JVM heap at 100ms intervals via a coroutine)

## Implementation

### 1. Asset: `benchmark_prompt.txt`
A single ~512-token English prompt — e.g. an excerpt from a public-domain technical article (~1700 words). One paragraph per topic so the model can sustain a long decode.

### 2. Worker
```kotlin
@HiltWorker
class ModelBenchmarkWorker @AssistedInject constructor(
    @Assisted context: Context,
    @Assisted params: WorkerParameters,
    @Named(LlmHelperQualifiers.LITE_RT_LM) private val helper: LlmModelHelper,
    private val benchmarkRepo: BenchmarkRepository,
    private val notifier: BenchmarkNotifier,
) : CoroutineWorker(context, params) {

    override suspend fun doWork(): Result {
        val modelPath = inputData.getString(KEY_MODEL_PATH) ?: return Result.failure()
        val modelId = inputData.getString(KEY_MODEL_ID) ?: return Result.failure()
        val temperature = inputData.getFloat(KEY_TEMPERATURE, 0.7f)
        val topK = inputData.getInt(KEY_TOP_K, 40)
        val maxTokens = inputData.getInt(KEY_MAX_TOKENS, 512)
        val trials = inputData.getInt(KEY_TRIALS, 3)

        runCatching { setForeground(notifier.foregroundInfo("Starting benchmark", 0)) }
            .onFailure { Timber.w(it, "setForeground denied; continuing without notif") }

        val prompt = applicationContext.assets.open("benchmark_prompt.txt")
            .bufferedReader().use { it.readText() }
        val request = ChatRequest(
            messages = listOf(ChatMessage(role = Role.USER, content = prompt)),
            generationParameters = GenerationParameters(
                temperature = temperature, topK = topK, maxTokens = maxTokens,
            ),
        )

        // Measure init
        val initStart = SystemClock.elapsedRealtime()
        helper.initialize(modelPath)
        val initTimeMs = SystemClock.elapsedRealtime() - initStart

        var prefillSum = 0.0; var decodeSum = 0.0; var peakMem = 0L

        for (trial in 1..trials) {
            setForegroundAsync(notifier.foregroundInfo("Trial $trial/$trials", trial * 100 / trials))
            val sampler = MemorySampler.start(applicationContext)
            val prefillStart = SystemClock.elapsedRealtime()
            var firstDeltaAt = -1L
            var tokenCount = 0
            val configHash = BenchmarkConfig(temperature, topK, maxTokens, trials).hash()

            helper.runInference(request, enableThinking = false).collect { token ->
                when (token) {
                    is StreamToken.Delta -> {
                        if (firstDeltaAt < 0) firstDeltaAt = SystemClock.elapsedRealtime()
                        tokenCount++
                    }
                    is StreamToken.Done -> {}
                    is StreamToken.Error -> throw IllegalStateException(token.message)
                }
            }
            val end = SystemClock.elapsedRealtime()
            sampler.stop()

            val prefillMs = firstDeltaAt - prefillStart
            val decodeMs = end - firstDeltaAt
            val inputTokens = (prompt.length / 4).coerceAtLeast(1)  // rough estimate
            val prefillTps = if (prefillMs > 0) inputTokens * 1000.0 / prefillMs else 0.0
            val decodeTps = if (decodeMs > 0) tokenCount * 1000.0 / decodeMs else 0.0
            prefillSum += prefillTps; decodeSum += decodeTps
            peakMem = maxOf(peakMem, sampler.peakBytes())
        }

        val result = BenchmarkResult(
            modelId = modelId,
            configHash = BenchmarkConfig(temperature, topK, maxTokens, trials).hash(),
            initTimeMs = initTimeMs,
            prefillTokPerSec = prefillSum / trials,
            decodeTokPerSec = decodeSum / trials,
            peakMemoryBytes = peakMem,
            createdAt = System.currentTimeMillis(),
        )
        benchmarkRepo.insert(result)
        return Result.success()
    }

    companion object {
        const val UNIQUE_NAME = "model_benchmark"
        const val KEY_MODEL_PATH = "model_path"
        const val KEY_MODEL_ID = "model_id"
        const val KEY_TEMPERATURE = "temperature"
        const val KEY_TOP_K = "top_k"
        const val KEY_MAX_TOKENS = "max_tokens"
        const val KEY_TRIALS = "trials"
    }
}
```

### 3. MemorySampler
```kotlin
class MemorySampler private constructor(private val ctx: Context) {
    private var peak: Long = 0
    private var job: Job? = null

    fun stop() { job?.cancel() }
    fun peakBytes(): Long = peak

    companion object {
        fun start(ctx: Context): MemorySampler {
            val s = MemorySampler(ctx)
            s.job = CoroutineScope(Dispatchers.Default).launch {
                while (isActive) {
                    val rt = Runtime.getRuntime()
                    val jvm = rt.totalMemory() - rt.freeMemory()
                    val native = android.os.Debug.getNativeHeapAllocatedSize()
                    s.peak = maxOf(s.peak, jvm + native)
                    delay(100)
                }
            }
            return s
        }
    }
}
```

### 4. BenchmarkScheduler
```kotlin
@Singleton
class BenchmarkScheduler @Inject constructor(
    @ApplicationContext private val context: Context,
) {
    fun enqueue(modelPath: String, modelId: String, config: BenchmarkConfig) {
        val request = OneTimeWorkRequestBuilder<ModelBenchmarkWorker>()
            .setInputData(workDataOf(
                ModelBenchmarkWorker.KEY_MODEL_PATH to modelPath,
                ModelBenchmarkWorker.KEY_MODEL_ID to modelId,
                ModelBenchmarkWorker.KEY_TEMPERATURE to config.temperature,
                ModelBenchmarkWorker.KEY_TOP_K to config.topK,
                ModelBenchmarkWorker.KEY_MAX_TOKENS to config.maxTokens,
                ModelBenchmarkWorker.KEY_TRIALS to config.trials,
            ))
            .setConstraints(
                Constraints.Builder()
                    .setRequiresBatteryNotLow(true)
                    .build()
            )
            .build()
        WorkManager.getInstance(context)
            .enqueueUniqueWork(
                ModelBenchmarkWorker.UNIQUE_NAME,
                ExistingWorkPolicy.REPLACE,
                request,
            )
    }
}
```
