The current config uses `domain-config` with IP entries that don't match IP ranges. `includeSubdomains` doesn't work for IP addresses. The fix: allow cleartext globally (local HTTP is needed for Ollama/LMStudio) while relying on remote servers (OpenAI/Anthropic) to enforce HTTPS at the server level.

Replace the entire file content:
```xml
<?xml version="1.0" encoding="utf-8"?>
<network-security-config>
    <base-config cleartextTrafficPermitted="true">
        <trust-anchors>
            <certificates src="system" />
        </trust-anchors>
    </base-config>
</network-security-config>
```

This removes the broken per-IP config. All HTTP connections are allowed. OpenAI/Anthropic/HuggingFace servers reject non-HTTPS connections at the server level. Local Ollama/LMStudio servers over LAN can now connect via HTTP.

private fun preloadLocalModel(filePath: String) {
    val modelName = filePath.substringAfterLast("/").removeSuffix(".gguf")
    _uiState.update { it.copy(isLoadingModel = true, loadingModelName = modelName, modelLoadError = null) }
    viewModelScope.launch(Dispatchers.Default) {
        try {
            val provider = localLlmProvider.get().configure(filePath)
            // Force loading by sending a small prompt and stopping
            val loaded = llamaEngine.loadModel(filePath)
            if (loaded) {
                _uiState.update { it.copy(isLoadingModel = false, loadingModelName = "") }
            } else {
                _uiState.update { it.copy(isLoadingModel = false, modelLoadError = "Failed to load model") }
            }
        } catch (e: Exception) {
            _uiState.update { it.copy(isLoadingModel = false, modelLoadError = e.message) }
        }
    }
}
```

Update `useEndpoint()`:

fun useEndpoint(endpoint: Endpoint) {
    val modelId = endpoint.modelId ?: return
    viewModelScope.launch {
        endpointRepository.activateEndpoint(endpoint.id)
        activeModelSelection.select(modelId, endpoint.apiType)
        // Fetch available models from the endpoint

private suspend fun fetchEndpointModels(endpoint: Endpoint) {
    val modelId = endpoint.modelId ?: return
    try {
        val provider = providerRouter.resolve(endpoint, modelId)
        val result = provider.listModels()
        result.onSuccess { models ->
            _uiState.update { it.copy(availableEndpointModels = models) }
        }.onFailure {
            _uiState.update { it.copy(error = "Failed to fetch models: ${it.message}") }
        }
    } catch (e: Exception) {
        _uiState.update { it.copy(error = e.message) }
    }
}
```
